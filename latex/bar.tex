\documentclass{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

\pgfplotsset{myplot/.style={
    height=15cm,
    xbar stacked,
    axis x line*=bottom,
    axis y line*=left,
    bar width=7pt,
    enlarge y limits=0.1,
    ytick=data,
    table/col sep=comma,
    yticklabel style={font=\small},
    legend style={at={(0.5,-0.1)}, anchor=north,legend columns=-1}
}}

\newcommand{\datapath}{data_processed/}
\newcommand{\datafile}{cora_mod.csv}

\edef\myfile{\datapath\datafile}
\edef\expandedname{\expandafter\detokenize\expandafter{\filename}}

\begin{document}
\begin{tikzpicture}
    \begin{axis}[
            title=\expandedname,
            myplot,
            yticklabels from table={\myfile}{category},
            xlabel={$Q$}
        ]

        % avg segment
        \addplot+[fill=blue!60]
        table[y expr=\coordindex, x=avg, col sep=comma] {\myfile};
        \addlegendentry{Avg};

        % maxdiff segment
        \addplot+[fill=blue!20]
        table[y expr=\coordindex, x=maxdiff, col sep=comma] {\myfile};
        \addlegendentry{Max};

    \end{axis}
\end{tikzpicture}

\end{document}
